flowchart TD
    classDef trigger fill:#fce4e4,stroke:#cc0000,stroke-width:2px,color:#000
    classDef agreeable fill:#d9ead3,stroke:#38761d,stroke-width:2px,color:#000
    classDef disagreeable fill:#fff2cc,stroke:#b45f06,stroke-width:2px,color:#000
    classDef decision fill:#c9daf8,stroke:#1155cc,stroke-width:2px,color:#000
    classDef action fill:#f3f3f3,stroke:#666666,stroke-width:2px,color:#000
    classDef endstate fill:#e0e0e0,stroke:#666666,stroke-width:2px,color:#000
    classDef database fill:#d9d2e9,stroke:#674ea7,stroke-width:2px,color:#000
    classDef notes fill:#ffffff,stroke:#ced4da,stroke-width:1px,color:#000

    %% Legend & General Notes Box
    subgraph LEGEND ["Legend & General Notes"]
        direction TB
        L_GREEN["Participant Generally Agreeable"]:::agreeable
        L_YELLOW["Participant Generally Disagreeable"]:::disagreeable
        L_TEXT["DEFINITIONS\n• Participant: Any person including a legal entity\n• RAND: Reasonable And Non-Discriminatory terms and\n  conditions including without compensation\n• LoA: Letter of Assurance\n• EP: Essential Patent Claims (LoAs not needed for non-essential claims)\n• WG Chair: The Working Group Chair or his delegate\n\nCALL FOR ESSENTIAL PATENT CLAIMS RULES\n• Call shall be made at every meeting\n• Call shall go out to all participants\n• Call shall only be for essential patent claims"]:::notes
    end
    style LEGEND fill:#f8f9fa,stroke:#adb5bd,stroke-width:1.5px,color:#000

    %% 1. Triggers & Inputs
    START(("START")):::trigger
    T1["WG Chair issues call for essential patent claims"]:::action
    T2(["Notice, claim, or response at any time from any source"]):::trigger
    T3(["Working Group incorporates part of another standard\n(including non-IEEE)"]):::trigger
    T4(["Working Group incorporates another standard in whole\nand unmodified (including non-IEEE)"]):::trigger
    T5(["Submitter makes special PatCom request"]):::trigger

    %% 2. Initial Processing
    START --> T1
    T1 -->|"Participant responds"| P1["Participant responds to the call for essential patent claims"]:::action
    P1 --> O1["WG Chair sends LoA, cover letter & makes draft standard available"]:::action
    T2 --> O1

    T3 --> D1{"LoA on file for the incorporated part?"}:::decision
    T4 --> END1(["No Further Action Needed"]):::endstate

    D1 -- "Yes" --> END1
    D1 -- "No" --> D3

    %% 3. Outreach Decisions
    O1 --> D2{"Potential EP holder responds to LoA request?"}:::decision
    D2 -- "No" --> D3{"Any reason to assume potential EP?"}:::decision
    D3 -- "No" --> END1
    D3 -- "Yes" --> PATCOM["PatCom Consideration"]:::action

    %% 4. Response Evaluation (Routing)
    D2 -- "Yes" --> RESP_ROUTER{"Evaluate Response Type"}:::decision

    %% Agreeable Flow (Green)
    RESP_ROUTER -->|"Generally Agreeable"| G1["Has/claims EPs & agrees to RAND"]:::agreeable
    RESP_ROUTER -->|"Generally Agreeable"| G2["Does not know but agrees to RAND if discovered later"]:::agreeable
    RESP_ROUTER -->|"Generally Agreeable"| G3["Agrees not to enforce essential patent claim(s)"]:::agreeable

    %% Disagreeable Flow (Yellow)
    RESP_ROUTER -->|"Generally Disagreeable"| Y1["Has specific EPs & does not agree to RAND"]:::disagreeable
    RESP_ROUTER -->|"Generally Disagreeable"| Y2["Does not know & does not agree to RAND if discovered later"]:::disagreeable
    RESP_ROUTER -->|"Generally Disagreeable"| Y3["Asserts EPs with no specifics & refuses RAND"]:::disagreeable

    %% Special Track
    T5 -.->|"Possibly w/o WG knowledge"| PATCOM

    %% 5A. Agreeable Processing & Database
    G1 --> V1["Authorized person returns LoA to IEEE staff with copy to WG Chair"]:::action
    G2 --> V1
    G3 --> V1
    
    V1 --> D4{"Materially complete and on approved LOA form?"}:::decision
    D4 -- "No" --> R1["Potential EP holder is reminded of requirement to use approved form"]:::action
    R1 --> V1
    
    D4 -- "Yes" --> DB[("Patent LoA Database")]:::database

    %% Database Access Outputs
    DB -->|"Written request to staff"| OUT1["Copy of LoA sent to requestor"]:::action
    DB -->|"Web Access"| OUT2["List of EP holders, LoA category, and patent number if appropriate.\nLoA is scanned into DB and accessible from the Web."]:::action

    %% 5B. Disagreeable Processing
    Y1 --> PATCOM
    Y2 --> PATCOM
    Y3 --> PATCOM
    PATCOM --> OUT3(["PatCom recommendation to the SASB for appropriate action"]):::endstate
